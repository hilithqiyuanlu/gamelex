cmake_minimum_required(VERSION 3.20)
project(GameLexProbe VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(JPEG REQUIRED)
add_library(gamelex_core core/src/core.cpp core/src/image.cpp core/src/network.cpp core/src/form_http.cpp)
target_include_directories(gamelex_core PUBLIC core/include vendor)
target_link_libraries(gamelex_core PUBLIC CURL::libcurl JPEG::JPEG)
target_compile_options(gamelex_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(gamelex-probe scripts/host_probe.cpp)
target_link_libraries(gamelex-probe PRIVATE gamelex_core)
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(core-tests PRIVATE gamelex_core)
enable_testing()
add_test(NAME core COMMAND core-tests)
add_executable(ui-tests tests/ui_tests.cpp)
target_include_directories(ui-tests PRIVATE core/include)
target_compile_options(ui-tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME ui COMMAND ui-tests)
add_executable(reading-tests tests/reading_tests.cpp)
target_include_directories(reading-tests PRIVATE core/include)
target_compile_options(reading-tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME reading COMMAND reading-tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME https-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/https_test.py $<TARGET_FILE:gamelex-probe> $<TARGET_FILE:core-tests>)
set_tests_properties(https-integration PROPERTIES TIMEOUT 40)
add_test(NAME deployment COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/deploy_test.py)
add_executable(form-tests tests/form_tests.cpp)
target_link_libraries(form-tests PRIVATE gamelex_core)
target_compile_options(form-tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME form COMMAND form-tests)
add_executable(form-http-probe tests/form_http_probe.cpp)
target_link_libraries(form-http-probe PRIVATE gamelex_core)
add_test(NAME form-https-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/form_https_test.py $<TARGET_FILE:form-http-probe>)
set_tests_properties(form-https-integration PROPERTIES TIMEOUT 40)
add_executable(group-reader-tests tests/group_reader_tests.cpp)
target_link_libraries(group-reader-tests PRIVATE gamelex_core)
target_compile_options(group-reader-tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME group-reader COMMAND group-reader-tests)
add_executable(overlay-flow-tests tests/overlay_flow_tests.cpp)
target_include_directories(overlay-flow-tests BEFORE PRIVATE tests/ui_harness)
target_link_libraries(overlay-flow-tests PRIVATE gamelex_core)
target_compile_options(overlay-flow-tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME overlay-flow COMMAND overlay-flow-tests)
add_executable(protocol-tests tests/protocol_tests.cpp)
target_link_libraries(protocol-tests PRIVATE gamelex_core)
add_test(NAME protocol COMMAND protocol-tests)
add_test(NAME distribution COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/distribution_test.py)
